public class CursorState {

    // 0.02f вЂ” РўР•РЎРўРћР’РћР• Р·РЅР°С‡РµРЅРёРµ (Р»Р°Рі 2-3 СЃРµРєСѓРЅРґС‹). Р”Р»СЏ РёРіСЂС‹: 0.25f-0.5f.
    public static float catchUpSpeed = 0.02f;

    public static float cursorYaw = 0.0f;
    public static float cursorPitch = 0.0f;
    public static float prevCursorYaw = 0.0f;
    public static float prevCursorPitch = 0.0f;

    // РЎС‡С‘С‚С‡РёРє РІС‹Р·РѕРІРѕРІ СЂРµРґРёСЂРµРєС‚РѕРІ РёР· РјРёРєСЃРёРЅРѕРІ (СЃРј. EntityPickMixin Рё
    // GameRendererPickMixin). Р Р°СЃС‚С‘С‚ -> РјРёРєСЃРёРЅС‹ Р¶РёРІС‹ Рё РґР°СЋС‚
    // РїРѕРєР°РґСЂРѕРІСѓСЋ С‚РѕС‡РЅРѕСЃС‚СЊ СЂРµР№РєР°СЃС‚Р°. РЎС‚РѕРёС‚ РЅР° РјРµСЃС‚Рµ -> РјРёРєСЃРёРЅС‹ РЅРµ
    // РІРЅРµРґСЂРёР»РёСЃСЊ, СЂР°Р±РѕС‚Р°РµС‚ СЃРѕР±С‹С‚РёР№РЅС‹Р№ override (С‚РѕР¶Рµ РѕРє).
    public static long redirectCalls = 0;

    private static boolean initialized = false;
    private static long lastLoggedRedirects = 0;

    private static final Logger LOGGER = LogManager.getLogger("AASMovement/CursorState");
    private static int debugTickCounter = 0;

    public static void tick(float realYaw, float realPitch) {
        prevCursorYaw = cursorYaw;
        prevCursorPitch = cursorPitch;

        debugTickCounter++;
        if (debugTickCounter % 20 == 0) {
            boolean alive = redirectCalls > lastLoggedRedirects;
            lastLoggedRedirects = redirectCalls;
            LOGGER.info("[AASMovement DEBUG] realYaw={}, cursorYaw={}, delta={}, mixin={}",
                    realYaw, cursorYaw,
                    Mth.wrapDegrees(realYaw - cursorYaw),
                    alive ? "ALIVE (" + redirectCalls + ")" : "DEAD");
        }

        if (!initialized) {
            cursorYaw = realYaw;
            cursorPitch = realPitch;
            prevCursorYaw = realYaw;
            prevCursorPitch = realPitch;
            initialized = true;
            return;
        }

        cursorYaw += Mth.wrapDegrees(realYaw - cursorYaw) * catchUpSpeed;
        cursorPitch = Mth.lerp(catchUpSpeed, cursorPitch,
                Mth.clamp(realPitch, -90.0f, 90.0f));
    }

    public static void reset() {
        initialized = false;
        debugTickCounter = 0;
        redirectCalls = 0;
        lastLoggedRedirects = 0;
    }

    // РРЅС‚РµСЂРїРѕР»СЏС†РёСЏ РїРѕ РєСЂР°С‚С‡Р°Р№С€РµР№ РґСѓРіРµ (С„РёРєСЃ РєСѓРІС‹СЂРєР° РЅР° +/-180)
    public static float getInterpolatedYaw(float partialTick) {
        return prevCursorYaw
                + Mth.wrapDegrees(cursorYaw - prevCursorYaw) * partialTick;
    }

    public static float getInterpolatedPitch(float partialTick) {
        return Mth.lerp(partialTick, prevCursorPitch, cursorPitch);
    }

    public static Vec3 getInterpolatedDirection(float partialTick) {
        return calculateViewVector(
                getInterpolatedPitch(partialTick),
                getInterpolatedYaw(partialTick));
    }

    // РљРѕРїРёСЏ С„РѕСЂРјСѓР»С‹ Entity#calculateViewVector
    private static Vec3 calculateViewVector(float pitch, float yaw) {
        float f = pitch * ((float) Math.PI / 180F);
        float f1 = -yaw * ((float) Math.PI / 180F);
        float f2 = Mth.cos(f1);
        float f3 = Mth.sin(f1);
        float f4 = Mth.cos(f);
        float f5 = Mth.sin(f);
        return new Vec3((double) (f3 * f4), (double) (-f5), (double) (f2 * f4));
    }
}

        AABB box = player.getBoundingBox()
                .expandTowards(dir.scale(reach))
                .inflate(1.0, 1.0, 1.0);

                player, eye, end, box,
                e -> !e.isSpectator() && e.isPickable(),
                maxSqr);

            mc.hitResult = entityHit;
            mc.crosshairPickEntity = entityHit.getEntity();

            mc.hitResult = blockHit;

        if (id.equals(VanillaGuiOverlay.CROSSHAIR.id())
                && mc.player != null && !mc.player.isCreative()) {
            event.setCanceled(true);
        }
    }

    // РљСЂРµСЃС‚РёРє СЂРёСЃСѓРµРј РІ RenderGuiEvent.Post вЂ” Post РѕС‚РјРµРЅС‘РЅРЅРѕРіРѕ РѕРІРµСЂР»РµСЏ
    // CROSSHAIR РЅРµ РІС‹Р·С‹РІР°РµС‚СЃСЏ РЅРёРєРѕРіРґР° (СЌС‚Рѕ СѓР¶Рµ РїРѕС‡РёРЅРµРЅРѕ, РєСЂРµСЃС‚РёРє РІРёРґРµРЅ).
    @SubscribeEvent
    public void onRenderCrosshair(RenderGuiEvent.Post event) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null) return;
        if (mc.player.isCreative()) return;

        float pt = event.getPartialTick();

        float camYaw = mc.player.yRotO
                + Mth.wrapDegrees(mc.player.getYRot() - mc.player.yRotO) * pt;
        float camPitch = Mth.lerp(pt, mc.player.xRotO, mc.player.getXRot());
        float curYaw = CursorState.getInterpolatedYaw(pt);
        float curPitch = CursorState.getInterpolatedPitch(pt);

        float deltaYaw = Mth.wrapDegrees(curYaw - camYaw);
        float deltaPitch = curPitch - camPitch;

        double halfFovRad = Math.toRadians(mc.options.fov().get() / 2.0);
        int w = mc.getWindow().getGuiScaledWidth();
        int h = mc.getWindow().getGuiScaledHeight();
        double pxPerRadian = (h / 2.0) / Math.tan(halfFovRad);

        int cx = w / 2 + (int) (Math.tan(Math.toRadians(deltaYaw)) * pxPerRadian);
        int cy = h / 2 + (int) (Math.tan(Math.toRadians(deltaPitch)) * pxPerRadian);

        cx = Mth.clamp(cx, 6, w - 6);
        cy = Mth.clamp(cy, 6, h - 6);

        GuiGraphics gui = event.getGuiGraphics();
        int size = 4;
        int t = 1;
        int color = 0xCCFFFFFF;
        gui.fill(cx - size, cy - t, cx + size, cy + t, color);
        gui.fill(cx - t, cy - size, cx + t, cy + size, color);
    }

    // РҐРѕС‚Р±Р°СЂ вЂ” Р±РµР· РёР·РјРµРЅРµРЅРёР№ (Post РѕРІРµСЂР»РµСЏ CHAT_PANEL, РѕРЅ РЅРµ РѕС‚РјРµРЅС‘РЅ)
    @SubscribeEvent
    public void onRenderGuiPost(RenderGuiOverlayEvent.Post event) {
        if (!event.getOverlay().id().equals(VanillaGuiOverlay.CHAT_PANEL.id())) return;

        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null) return;

        float pt = event.getPartialTick();
        float renderAlpha = Mth.lerp(pt, prevHotbarAlpha, hotbarAlpha);
        float renderSlide = Mth.lerp(pt, prevHotbarSlide, hotbarSlide);

        if (renderAlpha <= 0.001f) return;

        GuiGraphics gui = event.getGuiGraphics();
        int screenWidth = mc.getWindow().getGuiScaledWidth();
        int screenHeight = mc.getWindow().getGuiScaledHeight();

        int slotSize = 22;
        int gap = 4;
        int totalWidth = 9 * (slotSize + gap);
        int startX = (screenWidth - totalWidth) / 2;

        float baseY = (screenHeight + 40) - (renderSlide * 72);
        int alpha = (int) (renderAlpha * 200);
        int color = (alpha << 24);

        for (int i = 0; i < 9; i++) {
            int x = startX + i * (slotSize + gap);
            float currentH = Mth.lerp(pt, prevSlotH[i], slotH[i]);
            int y = (int) (baseY - currentH);

            gui.fill(x, y, x + slotSize, y + slotSize, color);

            ItemStack stack = mc.player.getInventory().getItem(i);
            if (!stack.isEmpty()) {
                gui.renderItem(stack, x + 3, y + 3);
                gui.renderItemDecorations(mc.font, stack, x + 3, y + 3);
            }
        }
    }
